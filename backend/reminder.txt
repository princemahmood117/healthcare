class Student {
    // using constructor, we initialize the parameter value to "this.name" while caliling the class 
    constructor(name, age) {
        console.log("this is only 'this' which is an empty object :", this, "\n");

        this.name = name;
        this.age = age;
        console.log("name came from the new Student() call :", name );
        console.log("name is set to this.name : ", this.name);
    }
}

const stu1 = new Student("Piyal", 20)
console.log("\n","this is an object from the Class Student : ", stu1, "\n");
